# One fuzzing program per file reader (milestone 8, part 8); built with MOTEUR_FUZZ (preset windows-fuzz).
set(_fuzz_targets map effect animation save ktx2 gltf table bindings image archive)
set(_index 1)
foreach(name IN LISTS _fuzz_targets)
    add_executable(fuzz_${name} fuzz.cpp)
    target_compile_definitions(fuzz_${name} PRIVATE MOTEUR_FUZZ_TARGET=${_index})
    target_include_directories(fuzz_${name} PRIVATE "${PROJECT_SOURCE_DIR}/src/moteur/private")
    target_link_libraries(fuzz_${name} PRIVATE moteur)
    moteur_enable_warnings(fuzz_${name})
    if(MSVC)
        target_compile_options(fuzz_${name} PRIVATE /fsanitize=fuzzer)
        target_link_options(fuzz_${name} PRIVATE /fsanitize=fuzzer)
    else()
        target_compile_options(fuzz_${name} PRIVATE -fsanitize=fuzzer)
        target_link_options(fuzz_${name} PRIVATE -fsanitize=fuzzer)
    endif()
    math(EXPR _index "${_index} + 1")
endforeach()
